function securityLog($message, $type = 'INFO') {
    $log_entry = date('Y-m-d H:i:s') . " | $type | " . $_SERVER['REMOTE_ADDR'] . " | " . $message . PHP_EOL;

}

function isLockedOut() {
    if (isset($_SESSION['login_attempts']) && isset($_SESSION['first_attempt_time'])) {
        if ($_SESSION['login_attempts'] >= MAX_LOGIN_ATTEMPTS) {
            $time_elapsed = time() - $_SESSION['first_attempt_time'];
            if ($time_elapsed < LOCKOUT_TIME) {
                return true;
            } else {
                // Reset attempts setelah lockout habis
                $_SESSION['login_attempts'] = 0;
                $_SESSION['first_attempt_time'] = null;
            }
        }
    }
    return false;
}

function isFileSafe($filename) {
    // Cek ekstensi
    $ext = strtolower(pathinfo($filename, PATHINFO_EXTENSION));
    if (!in_array($ext, ALLOWED_EXTENSIONS)) {
        return false;
    }
    
    // Cek nama file terlarang
    foreach (BLOCKED_FILES as $blocked) {
        if (stripos($filename, $blocked) !== false) {
            return false;
        }
    }
    
    // Cek konten untuk PHP malware
    if ($ext === 'php' || $ext === 'phtml') {
        $content = @file_get_contents($filename);
        if ($content) {
            $dangerous = ['eval', 'base64_decode', 'system', 'exec', 'shell_exec', 'popen', 'passthru', 'proc_open'];
            foreach ($dangerous as $func) {
                if (stripos($content, $func) !== false) {
                    return false;
                }
            }
        }
    }
    
    return true;
}

function scanDirectory($dir) {
    $dangerous_files = [];
    $files = scandir($dir);
    foreach ($files as $file) {
        if ($file === '.' || $file === '..') continue;
        $path = $dir . '/' . $file;
        if (is_file($path) && !isFileSafe($path)) {
            $dangerous_files[] = $file;
        }
    }
    return $dangerous_files;
}

function formatSizeUnits($bytes) {
    $units = ['B', 'KB', 'MB', 'GB', 'TB'];
    $i = 0;
    while ($bytes >= 1024 && $i < count($units) - 1) {
        $bytes /= 1024;
        $i++;
    }
    return round($bytes, 2) . ' ' . $units[$i];
}

function encodePath($path) {
    return str_replace(['/', '\\', '.', ':'], ['à¦•', 'à¦–', 'à¦—', 'à¦˜'], $path);
}

function decodePath($path) {
    return str_replace(['à¦•', 'à¦–', 'à¦—', 'à¦˜'], ['/', '\\', '.', ':'], $path);
}

function isPathSafe($path) {
    // Cegah directory traversal
    $real = realpath($path);
    $root = realpath(__DIR__);
    if ($real === false || strpos($real, $root) !== 0) {
        return false;
    }
    return true;
}
