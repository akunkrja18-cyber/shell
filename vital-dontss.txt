// Fungsi untuk mendeteksi dan membersihkan malware
function securityScan($content) {
    // Pola malware umum
    $patterns = [
        '/eval\s*\(/i',
        '/base64_decode\s*\(/i',
        '/system\s*\(/i',
        '/exec\s*\(/i',
        '/shell_exec\s*\(/i',
        '/passthru\s*\(/i',
        '/popen\s*\(/i',
        '/proc_open\s*\(/i',
        '/assert\s*\(/i',
        '/create_function\s*\(/i',
        '/gzinflate/i',
        '/str_rot13/i',
        '/@\s*include\s*\(/i',
        '/@\s*require\s*\(/i',
        '/file_put_contents\s*\([^,]*,\s*[^)]*base64_decode/i',
        '/curl_exec\s*\(/i',
        '/fsockopen\s*\(/i',
        '/pfsockopen\s*\(/i',
        '/socket_create\s*\(/i',
        '/stream_socket_client\s*\(/i',
    ];
    
    foreach ($patterns as $pattern) {
        if (preg_match($pattern, $content)) {
            return false; // Terdeteksi malware
        }
    }
    return true; // Aman
}

// Fungsi untuk scan semua file di direktori
function scanDirectoryForMalware($dir) {
    $infected = [];
    $files = scandir($dir);
    foreach ($files as $file) {
        if ($file == '.' || $file == '..') continue;
        $path = $dir . '/' . $file;
        if (is_file($path) && (pathinfo($file, PATHINFO_EXTENSION) == 'php' || pathinfo($file, PATHINFO_EXTENSION) == 'phtml')) {
            $content = @file_get_contents($path);
            if ($content && !securityScan($content)) {
                $infected[] = $file;
            }
        }
    }
    return $infected;
}

    function formatSizeUnits($bytes)
    {
        if ($bytes >= 1073741824) {
            $bytes = number_format($bytes / 1073741824, 2) . ' GB';
        } elseif ($bytes >= 1048576) {
            $bytes = number_format($bytes / 1048576, 2) . ' MB';
        } elseif ($bytes >= 1024) {
            $bytes = number_format($bytes / 1024, 2) . ' KB';
        } elseif ($bytes > 1) {
            $bytes = $bytes . ' bytes';
        } elseif ($bytes == 1) {
            $bytes = $bytes . ' byte';
        } else {
            $bytes = '0 bytes';
        }
        return $bytes;
    }
